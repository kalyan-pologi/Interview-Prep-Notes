Functions

function add(a: number, b: number): number {
  return a + b;
}

const multiply = (a: number, b: number): number => a * b;

// optional & default parameters
function greet(name: string, age?: number): string {
  return `Hello ${name}`;
}
function log(msg: string = "default"): void {
  console.log(msg);
}

// rest parameters
function sum(...numbers: number[]): number {
  return numbers.reduce((a, b) => a + b, 0);
}

Generics

function identity<T>(value: T): T {
  return value;
}
let result = identity<string>("hello");

interface Box<T> {
  value: T;
}
let box: Box<number> = { value: 123 };

class DataHolder<T> {
  private data: T;
  constructor(data: T) { this.data = data; }
  getData(): T { return this.data; }
}

Classes

class User {
  private id: number;
  public name: string;
  protected email: string;
  readonly created: Date;

  constructor(id: number, name: string) {
    this.id = id;
    this.name = name;
    this.created = new Date();
  }

  getName(): string { return this.name; }
}

class Admin extends User {
  role: string = "admin";
}

abstract class Animal {
  abstract makeSound(): void;

Utility Types

interface User {
  id: number;
  name: string;
  email: string;
}

type PartialUser = Partial<User>;         // all optional
type RequiredUser = Required<User>;       // all required
type ReadonlyUser = Readonly<User>;       // all readonly

Type Guards

function print(value: string | number) {
  if (typeof value === "string") {
    console.log(value.toUpperCase());
  } else {
    console.log(value.toFixed(2));
  }
}

if (error instanceof Error) {
  console.log(error.message);
}

function isString(value: any): value is string {
  return typeof value === "string";
}

if ("email" in user) {
  console.log(user.email);
}

Promises & Async

function fetchData(): Promise<string> {
  return Promise.resolve("data");
}

async function getData(): Promise<User> {
  const response = await fetch("/api/user");
  const data: User = await response.json();
  return data;
}

async function loadData<T>(url: string): Promise<T> {
  const response = await fetch(url);
  return await response.json();
}

// mapped types

Key Concepts

  • Type Safety, catches errors at compile time instead of at runtime
  • Inference, TypeScript automatically infers types when they are not explicitly annotated
  • Strict Mode, enabled with "strict": true, catches null/undefined access and other unsafe patterns
  • Non-null assertion, user!.name tells the compiler a value is not null even though its type says it could be, use sparingly since it disables a real safety check
  • Type vs Interface, prefer interface for object shapes since it supports declaration merging, prefer type for unions and other complex type compositions
