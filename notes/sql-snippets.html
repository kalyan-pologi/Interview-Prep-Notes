🗄️ SQL Code Snippets

Reusable SQL code snippets, question by question.

DDL Basics

Q: Create a table with a primary key, a not-null column, and a unique column.

CREATE TABLE student (
    student_id INT PRIMARY KEY AUTO_INCREMENT,
    name       VARCHAR(20) NOT NULL,
    major      VARCHAR(20) UNIQUE
);

Q: Add and then drop a column from an existing table.

ALTER TABLE student ADD gpa DECIMAL(3, 2);
ALTER TABLE student DROP COLUMN gpa;

Q: Drop a table entirely.

DROP TABLE student;

Q: Create a new table from an existing table's data.

CREATE TABLE newtable AS SELECT * FROM EmployeeInfo;

Basic CRUD

Q: Insert a full row, and insert a row specifying only some columns.

INSERT INTO student VALUES (1, 'kalyan', 'maths');
INSERT INTO student(student_id, name) VALUES (2, 'prabhas');

Q: Select all rows, sorted by two columns.

SELECT name, major FROM student ORDER BY name, major DESC;

Q: Limit results to the first N rows.

SELECT * FROM student LIMIT 2;

Q: Filter rows where a column matches one of several values.

SELECT * FROM student WHERE name IN ('kalyan', 'prabhas');

Q: Filter rows with a partial text match.

SELECT * FROM student WHERE name LIKE '%kal%';

Q: Get the distinct values of a column.

SELECT DISTINCT sex FROM employee;

Q: Update a row's value based on a condition.

UPDATE student SET name = 'poloji' WHERE name = 'kalyan';

Q: Delete a row based on a condition.

DELETE FROM student WHERE name = 'poloji';

Foreign Keys

Q: What happens to child rows when the referenced parent row is deleted, if the foreign key should null them out instead of blocking the delete?

FOREIGN KEY (mrg_id) REFERENCES student(student_id) ON DELETE SET NULL;

Q: Same question, but child rows should be deleted along with the parent.

FOREIGN KEY (mrg_id) REFERENCES student(student_id) ON DELETE CASCADE;

Aggregates

Q: Count the number of employees.

SELECT COUNT(emp_id) FROM employee;

Q: Find the minimum salary in each department.

SELECT dept, MIN(salary) FROM employee GROUP BY dept;

Q: Count how many employees are in each department.

SELECT DepartmentID, COUNT(EmployeeID) AS EmployeeCount
FROM Employee GROUP BY DepartmentID;

UNION vs. UNION ALL

Q: Combine two result sets and remove duplicate rows.

SELECT name AS first_name FROM employee
UNION
SELECT major FROM student;

Q: Combine two result sets keeping every row, including duplicates (faster since it skips the dedup step).

SELECT emp_name, dept FROM employee e1 WHERE e1.dept = 'HR'
UNION ALL
SELECT emp_name, dept FROM employee e2 WHERE e2.dept = 'HR';

Joins

Q: Get only rows that match in both tables.

SELECT e.emp_id, e.first_name
FROM employee e
JOIN student s ON e.emp_id = s.std_id;

Q: Get all rows from the left table, with nulls where there's no match on the right.

SELECT e.emp_id, e.first_name
FROM employee e
LEFT JOIN student s ON e.emp_id = s.std_id;

Q: Get all rows from the right table, with nulls where there's no match on the left.

SELECT e.emp_id, e.first_name
FROM employee e
RIGHT JOIN student s ON e.emp_id = s.std_id;

Q: List each employee alongside their supervisor's name, using the employee table joined to itself.

SELECT a.emp_id AS "Emp_ID", a.emp_name AS "Employee Name",
       b.emp_id AS "Supervisor ID", b.emp_name AS "Supervisor Name"
FROM employee a
JOIN employee b ON a.emp_supv = b.emp_id;

Q: Same self join, but also include employees who have no supervisor at all.

SELECT a.emp_id, a.emp_name, b.emp_id, b.emp_name
FROM employee a
LEFT JOIN employee b ON a.emp_supv = b.emp_id;

Subqueries

Q: Find employees who sold over 30000 to a single client.

SELECT first_name FROM employee
WHERE emp_id IN (
    SELECT emp_id FROM work_with WHERE total_sales > 30000
);

Q: Find employees earning above the company average salary.

SELECT name FROM Employee
WHERE Salary > (SELECT AVG(Salary) FROM Employee);

Q: Find employees not assigned to any department.

SELECT * FROM Employee
WHERE Dept_no NOT IN (SELECT Department_no FROM Department);

Q: Find employees in departments with fewer than 5 people.

SELECT * FROM Employee
WHERE DepartmentID IN (
    SELECT DepartmentID FROM Employee
    GROUP BY DepartmentID HAVING COUNT(EmployeeID) < 5
);

Q: Find employees who earn more than their own manager.

SELECT e.EmployeeID, e.Name, e.Salary
FROM Employee e
WHERE e.ManagerID IS NOT NULL
  AND e.Salary > (
      SELECT m.Salary FROM Employee m WHERE m.EmployeeID = e.ManagerID
  );

Nth Highest Salary

Five different ways to solve the same classic question, worth knowing more than one since interviewers often ask for an alternative approach as a follow-up.

Q: Find the 4th highest salary using LIMIT with an offset.

SELECT salary FROM employee ORDER BY salary DESC LIMIT 3, 1;

Q: Find the Nth highest salary using a correlated subquery, without a window function.

SELECT salary FROM EmployeePosition E1
WHERE N - 1 = (
    SELECT COUNT(DISTINCT E2.Salary)
    FROM EmployeePosition E2
    WHERE E2.Salary > E1.Salary
);

Q: Find the 2nd highest salary using RANK() (note: RANK() skips numbers after a tie, e.g. 1,2,2,4).

SELECT salary FROM (
    SELECT salary, RANK() OVER (ORDER BY Salary DESC) AS rnk FROM Employee
) ranked
WHERE rnk = 2;

Q: Find the 3rd and 4th highest salary using DENSE_RANK() (no gaps after a tie, e.g. 1,2,2,3).

SELECT * FROM (
    SELECT salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS rnk FROM Employee
) ranked
WHERE rnk IN (3, 4);

Q: Find the 3rd and 4th highest salary with a CTE using ROW_NUMBER(), where every row gets a unique rank even with ties.

WITH SalaryRank AS (
    SELECT Salary, ROW_NUMBER() OVER (ORDER BY Salary DESC) AS rnk FROM Employee
)
SELECT Salary FROM SalaryRank WHERE rnk IN (3, 4);

Duplicate Rows

Q: Find rows that are duplicated across a set of columns.

SELECT e_id, name, age, COUNT(*) AS dup_count
FROM dup_employee
GROUP BY e_id, name, age
HAVING COUNT(*) > 1;

Q: Delete duplicate rows but keep exactly one copy of each (name, email) pair.

DELETE FROM Employee
WHERE EmployeeID NOT IN (
    SELECT MIN(EmployeeID) FROM Employee GROUP BY name, email
);

Q: Get distinct rows without using the DISTINCT keyword.

SELECT * FROM employee a
WHERE rowid = (
    SELECT MAX(rowid) FROM employee b
    WHERE a.employee_no = b.employee_no
);

Q: Remove duplicate rows using rowid, keeping the latest one per key.

DELETE FROM student a
WHERE rowid <> (
    SELECT MAX(rowid) FROM student b WHERE a.rollno = b.rollno
);

CASE Statement

Q: Classify each employee's salary into High, Medium, or Low.

SELECT ID,
    CASE
        WHEN salary > 10000 THEN 'High'
        WHEN salary BETWEEN 5000 AND 10000 THEN 'Medium'
        ELSE 'Low'
    END AS SalaryLevel
FROM Employees;

Q: Check whether a table has any rows at all.

SELECT
    CASE WHEN EXISTS (SELECT * FROM TableName) THEN 'Has Records'
         ELSE 'No Records'
    END AS Status;

String Functions

Q: Convert a name to uppercase.

SELECT UPPER(EmpFname) AS EmpName FROM EmployeeInfo;

Q: Get the first 5 characters of a name.

SELECT SUBSTRING(student_name, 1, 5) FROM student;

Q: Get the last 5 characters of a name.

SELECT RIGHT(student_name, 5) FROM student;

Q: Replace spaces in a name with hyphens.

SELECT REPLACE(FullName, ' ', '-') FROM EmployeeDetails;

Q: Find the position of a character within a string, case-sensitively.

SELECT INSTR(FIRST_NAME, BINARY 'a') FROM Worker WHERE FIRST_NAME = 'Amitabh';

Date Functions

Q: Find employees who joined in the last 30 days.

SELECT * FROM Employee WHERE JoinDate > DATE_SUB(CURDATE(), INTERVAL 30 DAY);
-- equivalent
SELECT * FROM Employee WHERE JoinDate >= CURRENT_DATE - INTERVAL 30 DAY;

Q: Check whether a string is a valid date.

SELECT ISDATE('01/04/2020') AS "MM/DD/YY";  -- returns 1 (true) or 0 (false)

Miscellaneous Queries

Q: Fetch the first 50% of records in a table.

SELECT * FROM EmployeeInfo
WHERE EmpId <= (SELECT COUNT(EmpId) / 2 FROM EmployeeInfo);

Q: Find names that both start and end with "A".

SELECT name FROM Employee WHERE name LIKE 'A%' AND name LIKE '%A';

Q: Find employees who have no manager.

SELECT * FROM Employee WHERE ManagerID IS NULL;

Q: Get all records ordered by department ascending, then last name descending.

SELECT * FROM EmployeeInfo ORDER BY Department ASC, EmpLname DESC;

Q: Find records with matching ids across two different tables.

SELECT * FROM EmployeeSalary
WHERE EmpId IN (SELECT EmpId FROM ManagerSalary);

Q: Get the first record and the last record, by id.

SELECT * FROM EmployeeInfo WHERE EmpID = (SELECT MIN(EmpID) FROM EmployeeInfo);
SELECT * FROM EmployeeInfo WHERE EmpID = (SELECT MAX(EmpID) FROM EmployeeInfo);

Q: Find employees who also hold a managerial position.

SELECT E.EmpFname, E.EmpLname, P.EmpPosition
FROM EmployeeInfo E
INNER JOIN EmployeePosition P ON E.EmpID = P.EmpID
AND P.EmpPosition = 'Manager';

Q: Get total salary paid out per position.

SELECT EmpPosition, SUM(Salary) FROM EmployeePosition GROUP BY EmpPosition;

Q: Get the top N highest paid employees.

SELECT * FROM EmployeeInfo ORDER BY salary DESC LIMIT 5;

Q: Return only the even-numbered rows, using a window function.

SELECT customerId FROM (
    SELECT ROW_NUMBER() OVER () AS rowno, customerId FROM customers
) t
WHERE MOD(rowno, 2) = 0;

Stored Procedure

Q: Write a reusable stored procedure that inserts a new user given just a name.

DELIMITER $$
CREATE PROCEDURE AddUser(IN userName VARCHAR(100))
BEGIN
    INSERT INTO Users(name) VALUES(userName);
END $$
DELIMITER ;
